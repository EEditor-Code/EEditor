set(TARGET_NAME LibCommon)

file(GLOB_RECURSE EE_CORE_SRC CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
file(GLOB_RECURSE EE_CORE_HDR CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.h")

if(EE_BUILD_SHARED)
    add_library(${TARGET_NAME} SHARED ${EE_CORE_SRC} ${EE_CORE_HDR})
    target_compile_definitions(${TARGET_NAME} PRIVATE LIB_COMMON_EXPORT)
else()
    add_library(${TARGET_NAME} STATIC ${EE_CORE_SRC} ${EE_CORE_HDR})
endif()

target_include_directories(${TARGET_NAME}
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
        $<INSTALL_INTERFACE:include/${TARGET_NAME}>)

if(MSVC)
    target_compile_options(${TARGET_NAME} PRIVATE /utf-8 /W3)
    target_compile_definitions(${TARGET_NAME} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()

set_target_properties(${TARGET_NAME} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

install(TARGETS ${TARGET_NAME}
    RUNTIME DESTINATION bin
    ARCHIVE DESTINATION lib
    LIBRARY DESTINATION lib)

install(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/"
    DESTINATION include/${TARGET_NAME}
    FILES_MATCHING PATTERN "*.h")
